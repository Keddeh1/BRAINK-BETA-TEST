def unpack_and_audit_context(self, frame: bytes) -&gt; Tuple[bool, str, dict]:
    if len(frame) &lt; self.SIGNED_CONTEXT_SIZE:
        return False, "PACKET_UNDERSIZED", {}

    magic = frame[0:4]
    epoch = struct.unpack("&gt;Q", frame[4:12])[0]
    arch = frame[12:16]
    length = struct.unpack("&gt;I", frame[16:20])[0]
    target_hmac = frame[20:52]

    if magic != self.MAGIC_TOKEN:
        return False, "BAD_MAGIC_TOKEN", {}
    if epoch &lt;= self.local_epoch_floor:
        return False, "EPOCH_DOWNGRADE_ATTEMPT", {}
    if arch != self.expected_arch:
        return False, "ARCHITECTURE_MISMATCH", {}

    return True, "NOMINAL", {"epoch": epoch, "arch": arch, "length": length, "target_hmac": target_hmac}
